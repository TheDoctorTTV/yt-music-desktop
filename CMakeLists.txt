cmake_minimum_required(VERSION 3.24)
project(yt_music_desktop VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets WebEngineWidgets Network DBus)
qt_add_executable(youtube-music-desktop src/main.cpp)
target_compile_definitions(youtube-music-desktop PRIVATE YTMUSIC_VERSION="${PROJECT_VERSION}")
qt_add_resources(youtube-music-desktop assets PREFIX / FILES src/player.js icons/youtubemusic.png)
target_link_libraries(youtube-music-desktop PRIVATE Qt6::Widgets Qt6::WebEngineWidgets Qt6::Network Qt6::DBus)
target_compile_options(youtube-music-desktop PRIVATE -Wall -Wextra -Wpedantic)
include(GNUInstallDirs)
install(TARGETS youtube-music-desktop RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/net.thedoctorttv.ytmusicdesktop.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES icons/youtubemusic.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/512x512/apps RENAME net.thedoctorttv.ytmusicdesktop.png)
include(CTest)
if(BUILD_TESTING)
  find_program(NODE_EXECUTABLE node)
  if(NODE_EXECUTABLE)
    add_test(NAME player-bridge COMMAND ${NODE_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/player.test.cjs)
  endif()
endif()
